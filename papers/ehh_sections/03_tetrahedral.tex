\section{The tetrahedron as a minimal Pauli/SIC frame}

Define
\[
\mathbf n_1=\frac{(1,1,1)}{\sqrt3},
\quad
\mathbf n_2=\frac{(1,-1,-1)}{\sqrt3},
\quad
\mathbf n_3=\frac{(-1,1,-1)}{\sqrt3},
\quad
\mathbf n_4=\frac{(-1,-1,1)}{\sqrt3}.
\]
They satisfy
\[
\sum_a\mathbf n_a=0,
\qquad
\mathbf n_a\cdot\mathbf n_b=-\frac13\quad(a\ne b),
\qquad
\frac14\sum_a\mathbf n_a\mathbf n_a^T=\frac13I_3.
\]

Associate rank-one projectors
\[
P_a=\frac12(I+\mathbf n_a\cdot\boldsymbol\sigma).
\]
Then
\[
\boxed{
\sum_{a=1}^4P_a=2I,
\qquad
\operatorname{Tr}(P_aP_b)=\frac13\quad(a\ne b).
}
\]
Therefore $E_a=P_a/2$ is the tetrahedral qubit SIC-POVM. Four outcomes are minimal for informational completeness in dimension two because the real vector space of Hermitian $2\times2$ matrices has dimension four.

\begin{theorem}[Pauli orbit theorem]\label{thm:pauli-orbit}
The four tetrahedral projectors are exactly the projective Pauli orbit of $P_1$:
\[
\boxed{
\{P_1,P_2,P_3,P_4\}
=
\{UP_1U^\dagger:
U\in\{I,\sigma_x,\sigma_y,\sigma_z\}\}.
}
\]
Projectively the acting group is $V_4$. Its determinant-one spin lift is
\[
Q_8=
\{\pm I,\pm i\sigma_x,\pm i\sigma_y,\pm i\sigma_z\}.
\]
\end{theorem}

\begin{proof}
Conjugation by $\sigma_x$ preserves the $x$ Bloch component and flips $y,z$; cyclically for the other Pauli matrices. Acting on $(1,1,1)/\sqrt3$ therefore produces the four displayed sign patterns.
\end{proof}

This provides the precise reason the tetrahedron is natural here: it is simultaneously the minimal qubit SIC frame and the orbit of the same $Q_8/V_4$ Pauli structure already present in the spinorial lift.

\section{Binary-tetrahedral completion}

Define the order-three spin rotation
\[
U_3
=
-\frac12 I
-\frac{i}{2}
(\sigma_x+\sigma_y+\sigma_z).
\]
A direct Pauli calculation gives
\[
U_3^3=I,
\]
and conjugation cyclically permutes the Pauli axes:
\[
\sigma_x\mapsto\sigma_z,
\qquad
\sigma_z\mapsto\sigma_y,
\qquad
\sigma_y\mapsto\sigma_x.
\]

\begin{theorem}[Binary-tetrahedral completion]\label{thm:binary-tetra}
The subgroup of $SU(2)$ generated by $Q_8$ and $U_3$ is the binary tetrahedral group $2T$ of order $24$. It satisfies
\[
\boxed{
2T\cong Q_8\rtimes C_3,
\qquad
2T/\{\pm I\}\cong A_4.
}
\]
Thus $Q_8$ is the normal quaternionic core of the full orientation-preserving spin symmetry of the tetrahedron.
\end{theorem}

Projectively, $V_4$ gives the four-point orbit and the additional threefold rotation supplies the stabilizer/cycle structure needed to generate $A_4$.

\section{Tetrahedral null cone and spinor factorization}

With Minkowski metric $\eta=\operatorname{diag}(+,-,-,-)$ define
\[
k_a=(1,\mathbf n_a).
\]
Each $k_a$ is future null. For $p_a\ge0$ set
\[
P=\sum_ap_ak_a.
\]

\begin{theorem}[Exact tetrahedral causal norm]\label{thm:cone}
\[
\boxed{
P^2
=
\frac83\sum_{a<b}p_ap_b
\ge0.
}
\]
Therefore the positive span of the four null rays lies inside the future causal cone, with nullity only on the generating rays.
\end{theorem}

Each null vector has the standard rank-one spinor factorization
\[
k_{a,\alpha\dot\alpha}
=
\lambda_{a,\alpha}
\overline{\lambda}_{a,\dot\alpha}.
\]
The phase gauge is
\[
\lambda_a\sim e^{i\chi_a}\lambda_a.
\]

\section{Equivariant minimal Naimark dilation}

Choose normalized spinors $\xi_a\in\mathbb C^2$ such that
\[
P_a=|\xi_a\rangle\langle\xi_a|,
\qquad
E_a=\frac12P_a.
\]
Set
\[
\phi_a=\frac1{\sqrt2}\xi_a,
\qquad
E_a=|\phi_a\rangle\langle\phi_a|.
\]
Since $\sum_aE_a=I$, the analysis map
\[
\boxed{
V:\mathbb C^2\longrightarrow\mathbb C^4,
\qquad
(V\psi)_a=\langle\phi_a,\psi\rangle
}
\]
satisfies
\[
V^\dagger V=I_2.
\]
With the canonical outcome projectors
\[
\Pi_a=|e_a\rangle\langle e_a|
\]
on $\mathbb C^4$,
\[
\boxed{E_a=V^\dagger\Pi_aV.}
\]

\begin{theorem}[Minimal tetrahedral Naimark dilation]\label{thm:naimark}
The dilation $(\mathbb C^4,V,\{\Pi_a\})$ is minimal. Consequently its unitary-equivalence class is fixed by the labeled tetrahedral SIC.
\end{theorem}

\begin{proof}
For each outcome,
\[
\Pi_aV\phi_a
=
\langle\phi_a,\phi_a\rangle e_a
=
\frac12e_a.
\]
Hence every canonical basis vector belongs to the cyclic span generated by the compressed measurement, so that span is all of $\mathbb C^4$.
\end{proof}

A phase change $\xi_a\mapsto e^{i\theta_a}\xi_a$ changes the analysis map only by a diagonal unitary on the outcome carrier.

For $g\in2T$, write
\[
g\xi_a
=
e^{i\chi_{g,a}}\xi_{\pi_g(a)},
\qquad
\pi_g\in A_4.
\]
Define
\[
\boxed{
R(g)e_a
=
e^{i\chi_{g,a}}e_{\pi_g(a)}.
}
\]
The phase cocycle inherited from the spin representation gives $R(gh)=R(g)R(h)$.

\begin{theorem}[$2T$-equivariant Naimark dilation]\label{thm:naimark-equiv}
\[
\boxed{
Vg=R(g)V.
}
\]
Thus the minimal tetrahedral Naimark dilation carries an explicit monomial representation of the binary tetrahedral symmetry.
\end{theorem}

Covariant Naimark dilations of group-covariant POVMs are standard; see, for example, \cite{Decker2005}. The point here is the explicit tetrahedral specialization used below.

\section{Fermionic second quantization and the CAR carrier}

The spinor coefficient in the supercharge transforms with phase $e^{i\chi_{g,a}}$. Therefore use the conjugate outcome representation
\[
\overline{R}(g)e_a
=
e^{-i\chi_{g,a}}e_{\pi_g(a)}
\]
on the one-particle fermion carrier.

Form the antisymmetric Fock space
\[
\boxed{
\mathcal F_-(\mathbb C^4)
=
\Lambda^\bullet\mathbb C^4.
}
\]
For the canonical outcome basis define
\[
f_a^\dagger=c^\dagger(e_a),
\qquad
f_a=c(e_a).
\]
Standard fermionic second quantization gives
\[
\boxed{
\{f_a,f_b^\dagger\}=\delta_{ab},
\qquad
\{f_a,f_b\}=\{f_a^\dagger,f_b^\dagger\}=0.
}
\]
Hence, once the standard fermionic Fock functor is chosen, the four CAR modes are induced by the minimal SIC outcome carrier rather than independently selected.

Let $z=-I_2$ be the center of $2T$. Since $z\xi_a=-\xi_a$,
\[
R(z)=\overline R(z)=-I_4.
\]
Second quantization therefore yields
\[
\boxed{
\Gamma_-(\overline R(-I_2))
=
\bigoplus_{n=0}^4(-1)^n I_{\Lambda^n\mathbb C^4}
=
(-1)^F.
}
\]

\begin{theorem}[Spin-center / fermion-parity correspondence]\label{thm:center-parity}
For the tetrahedral SIC minimal Naimark carrier,
\[
\boxed{
-I_{\rm spinor}
\longmapsto
-I_{\rm outcome}
\longmapsto
(-1)^F.
}
\]
\end{theorem}

This is an algebraic representation statement, not the physical spin-statistics theorem. The fact that the canonical CAR grading is implemented by second quantization of the one-particle sign is standard CAR/Fock theory; cf. \cite{BratteliRobinson1997}.

\section{Tetrahedral $\mathcal N=1$ supertranslation algebra}

Normalize the null spinors by
\[
\lambda_a=\sqrt2\,\xi_a,
\]
so that
\[
\lambda_a\lambda_a^\dagger
=
I+\mathbf n_a\cdot\boldsymbol\sigma.
\]
For nonnegative weights $p_a$ define
\[
P_{\alpha\dot\beta}
=
\sum_a
p_a
\lambda_{a,\alpha}
\overline{\lambda}_{a,\dot\beta},
\]
and
\[
Q_\alpha
=
\sqrt2
\sum_a
\sqrt{p_a}\,
\lambda_{a,\alpha}f_a^\dagger.
\]

\begin{theorem}[Tetrahedral supertranslation algebra]\label{thm:tetra-susy}
\[
\boxed{
\{Q_\alpha,\overline Q_{\dot\beta}\}
=
2P_{\alpha\dot\beta},
}
\]
while
\[
\{Q_\alpha,Q_\beta\}=0,
\qquad
\{\overline Q_{\dot\alpha},\overline Q_{\dot\beta}\}=0.
\]
Moreover
\[
\boxed{
(-1)^FQ_\alpha(-1)^F=-Q_\alpha.
}
\]
Hence the tetrahedral null frame admits an exact $3+1$ dimensional $\mathcal N=1$ supertranslation realization after standard fermionic second quantization of the minimal SIC outcome carrier.
\end{theorem}

\begin{proof}
The mixed anticommutator collapses by the CAR:
\[
\{Q_\alpha,\overline Q_{\dot\beta}\}
=
2
\sum_a
p_a
\lambda_{a,\alpha}
\overline{\lambda}_{a,\dot\beta}.
\]
Same-chirality anticommutators vanish, and each creation operator is odd under fermion parity.
\end{proof}

\section{Binary-tetrahedral covariance of the induced supercharge}

Fermionic second quantization of the conjugate outcome representation satisfies
\[
\Gamma_-(\overline R(g))
f_a^\dagger
\Gamma_-(\overline R(g))^\dagger
=
e^{-i\chi_{g,a}}
f_{\pi_g(a)}^\dagger.
\]
The outcome phase therefore cancels the spinor phase.

\begin{theorem}[Binary-tetrahedral equivariance]\label{thm:tetra-equiv}
Under the simultaneous spin/Fock action,
\[
\boxed{
\bigl(g,\Gamma_-(\overline R(g))\bigr):
Q(p)\longmapsto Q(g\cdot p),
}
\]
where $(g\cdot p)_{\pi_g(a)}=p_a$. For equal tetrahedral weights the supercharge is invariant under the combined $2T$ action.

For the central element $-I_2$, the spinor factor and fermion-parity factor each contribute a minus sign, so their combined action fixes the supercharge.
\end{theorem}

\subsection{Equal-weight spinorial rest frame}

For equal tetrahedral weights $p_a=p$,
\[
P^\mu
=
p\sum_{a=1}^4 k_a^\mu
=
(4p,0,0,0),
\]
and therefore
\[
\boxed{P^2=16p^2.}
\]
The projective tetrahedral group $A_4$ fixes this timelike axis and permutes the four null generators. Its spin lift lies in
\[
2T\subset SU(2)=\operatorname{Spin}(3)
\subset
SL(2,\mathbb C)=\operatorname{Spin}^+(1,3).
\]
Hence the equal-weight supertranslation frame is an exact discrete spinorial rest frame: the momentum is fixed while the null decomposition is permuted. With the compensating CAR action of Theorem~\ref{thm:tetra-equiv}, the equal-weight supercharge tensor is invariant under the combined $2T$ action.
