\section{The tetrahedron as a minimal Pauli/SIC frame}

Define
\[
\mathbf n_1=\frac{(1,1,1)}{\sqrt3},
\quad
\mathbf n_2=\frac{(1,-1,-1)}{\sqrt3},
\quad
\mathbf n_3=\frac{(-1,1,-1)}{\sqrt3},
\quad
\mathbf n_4=\frac{(-1,-1,1)}{\sqrt3}.
\]
They satisfy
\[
\sum_a\mathbf n_a=0,
\qquad
\mathbf n_a\cdot\mathbf n_b=-\frac13\quad(a\ne b),
\qquad
\frac14\sum_a\mathbf n_a\mathbf n_a^T=\frac13I_3.
\]

Associate rank-one projectors
\[
P_a=\frac12(I+\mathbf n_a\cdot\boldsymbol\sigma).
\]
Then
\[
\boxed{
\sum_{a=1}^4P_a=2I,
\qquad
\operatorname{Tr}(P_aP_b)=\frac13\quad(a\ne b).
}
\]
Therefore $E_a=P_a/2$ is the tetrahedral qubit SIC-POVM. Four outcomes are minimal for informational completeness in dimension two because the real vector space of Hermitian $2\times2$ matrices has dimension four.

\begin{theorem}[Pauli orbit theorem]\label{thm:pauli-orbit}
The four tetrahedral projectors are exactly the projective Pauli orbit of $P_1$:
\[
\boxed{
\{P_1,P_2,P_3,P_4\}
=
\{UP_1U^\dagger:
U\in\{I,\sigma_x,\sigma_y,\sigma_z\}\}.
}
\]
Projectively the acting group is $V_4$. Its determinant-one spin lift is
\[
Q_8=
\{\pm I,\pm i\sigma_x,\pm i\sigma_y,\pm i\sigma_z\}.
\]
\end{theorem}

\begin{proof}
Conjugation by $\sigma_x$ preserves the $x$ Bloch component and flips $y,z$; cyclically for the other Pauli matrices. Acting on $(1,1,1)/\sqrt3$ therefore produces the four displayed sign patterns.
\end{proof}

This provides the precise reason the tetrahedron is natural here: it is simultaneously the minimal qubit SIC frame and the orbit of the same $Q_8/V_4$ Pauli structure already present in the spinorial lift.

\section{Binary-tetrahedral completion}

Define the order-three spin rotation
\[
U_3
=
-\frac12 I
-\frac{i}{2}
(\sigma_x+\sigma_y+\sigma_z).
\]
A direct Pauli calculation gives
\[
U_3^3=I,
\]
and conjugation cyclically permutes the Pauli axes:
\[
\sigma_x\mapsto\sigma_z,
\qquad
\sigma_z\mapsto\sigma_y,
\qquad
\sigma_y\mapsto\sigma_x.
\]

\begin{theorem}[Binary-tetrahedral completion]\label{thm:binary-tetra}
The subgroup of $SU(2)$ generated by $Q_8$ and $U_3$ is the binary tetrahedral group $2T$ of order $24$. It satisfies
\[
\boxed{
2T\cong Q_8\rtimes C_3,
\qquad
2T/\{\pm I\}\cong A_4.
}
\]
Thus $Q_8$ is the normal quaternionic core of the full orientation-preserving spin symmetry of the tetrahedron.
\end{theorem}

Projectively, $V_4$ gives the four-point orbit and the additional threefold rotation supplies the stabilizer/cycle structure needed to generate $A_4$.

\section{Tetrahedral null cone and spinor factorization}

With Minkowski metric $\eta=\operatorname{diag}(+,-,-,-)$ define
\[
k_a=(1,\mathbf n_a).
\]
Each $k_a$ is future null. For $p_a\ge0$ set
\[
P=\sum_ap_ak_a.
\]

\begin{theorem}[Exact tetrahedral causal norm]\label{thm:cone}
\[
\boxed{
P^2
=
\frac83\sum_{a<b}p_ap_b
\ge0.
}
\]
Therefore the positive span of the four null rays lies inside the future causal cone, with nullity only on the generating rays.
\end{theorem}

Each null vector has the standard rank-one spinor factorization
\[
k_{a,\alpha\dot\alpha}
=
\lambda_{a,\alpha}
\overline{\lambda}_{a,\dot\alpha}.
\]
The phase gauge is
\[
\lambda_a\sim e^{i\chi_a}\lambda_a.
\]

\section{Tetrahedral $\mathcal N=1$ supertranslation algebra}

Attach fermionic modes satisfying
\[
\{f_a,f_b^\dagger\}=\delta_{ab},
\qquad
\{f_a,f_b\}=\{f_a^\dagger,f_b^\dagger\}=0.
\]
Define
\[
P_{\alpha\dot\beta}
=
\sum_a
p_a
\lambda_{a,\alpha}
\overline{\lambda}_{a,\dot\beta},
\]
and
\[
Q_\alpha
=
\sqrt2
\sum_a
\sqrt{p_a}\,
\lambda_{a,\alpha}f_a^\dagger.
\]

\begin{theorem}[Tetrahedral supertranslation algebra]\label{thm:tetra-susy}
\[
\boxed{
\{Q_\alpha,\overline Q_{\dot\beta}\}
=
2P_{\alpha\dot\beta},
}
\]
while
\[
\{Q_\alpha,Q_\beta\}=0,
\qquad
\{\overline Q_{\dot\alpha},\overline Q_{\dot\beta}\}=0.
\]
Hence the tetrahedral null frame admits an exact $3+1$ dimensional $\mathcal N=1$ supertranslation realization once one CAR mode is supplied for each ray.
\end{theorem}

\begin{proof}
The mixed anticommutator collapses by $\{f_a,f_b^\dagger\}=\delta_{ab}$:
\[
\{Q_\alpha,\overline Q_{\dot\beta}\}
=
2
\sum_a
p_a
\lambda_{a,\alpha}
\overline{\lambda}_{a,\dot\beta}.
\]
Same-chirality anticommutators vanish by the CAR.
\end{proof}

\section{Binary-tetrahedral covariance of the supercharge frame}

For $g\in2T$, the spin action permutes tetrahedral rays projectively. Thus there are phases $\chi_{g,a}$ and a permutation $\pi_g\in A_4$ such that
\[
g\lambda_a
=
e^{i\chi_{g,a}}
\lambda_{\pi_g(a)}.
\]
Because the CAR algebra is invariant under unitary basis changes, choose a Fock-space lift $V_g$ with
\[
V_g f_a^\dagger V_g^{-1}
=
e^{-i\chi_{g,a}}
f_{\pi_g(a)}^\dagger.
\]

Write the geometric supercharge as the tensor
\[
\mathcal Q(p)
=
\sqrt2
\sum_a
\sqrt{p_a}\,
\lambda_a\otimes f_a^\dagger.
\]

\begin{theorem}[Binary-tetrahedral equivariance]\label{thm:tetra-equiv}
With the simultaneous spin/Fock action,
\[
\boxed{
(g\otimes V_g)\mathcal Q(p)
=
\mathcal Q(\pi_gp),
}
\]
where the weight vector is permuted with the rays. For equal tetrahedral weights the supercharge tensor is invariant under the combined $2T$ action, and its supertranslation anticommutator is covariant under the corresponding $A_4$ rotation.
\end{theorem}

The covariance theorem does not derive the fermionic CAR algebra from geometry; it states that once the CAR sector is supplied, the supersymmetry construction respects the full spinorial symmetry of the tetrahedral frame.

\subsection{Equal-weight spinorial rest frame}

For equal tetrahedral weights $p_a=p$,
\[
P^\mu
=
p\sum_{a=1}^4 k_a^\mu
=
(4p,0,0,0),
\]
and therefore
\[
\boxed{P^2=16p^2.}
\]
The projective tetrahedral group $A_4$ fixes this timelike axis and permutes the four null generators. Its spin lift lies in
\[
2T\subset SU(2)=\operatorname{Spin}(3)
\subset
SL(2,\mathbb C)=\operatorname{Spin}^+(1,3).
\]
Hence the equal-weight supertranslation frame is an exact discrete spinorial rest frame: the momentum is fixed while the null decomposition is permuted. With the compensating CAR action of Theorem~\ref{thm:tetra-equiv}, the equal-weight supercharge tensor is invariant under the combined $2T$ action.
